<!-- DemoStoreProviderファイルの内容をuser情報とサービス層に分割。API想定 -->
/* user情報 */
import { type PropsWithChildren, useState } from 'react';
import { DemoStoreContext } from '@/app/demo-store-context';
import type { SessionUser } from '@/domain/models';
import { login as loginApi } from '@/api/products';

export function DemoStoreProvider({ children }: PropsWithChildren) {
  const [activeUser, setActiveUser] = useState<SessionUser | null>(null);
  const login = async (email: string, password: string) => {
    const sessionUser = await loginApi(email, password);
    setActiveUser(sessionUser);
  };
  return (
    <DemoStoreContext.Provider
      value={{
        activeUser,
        login,
      }}
    >
      {children}
    </DemoStoreContext.Provider>
  );
}

/* サービス層 */
import { type PropsWithChildren, useState } from 'react';
import { ServiceContext } from '@/app/service-context';
import type { CreateProductInput } from '@/domain/models';

export function ServiceProvider({ children }: PropsWithChildren) {
  const [listingDraft, setListingDraft] =
    useState<CreateProductInput | null>(null);

  const saveListingDraft = (draft: CreateProductInput | null) => {
    setListingDraft(draft);
  };

  return (
    <ServiceContext.Provider
      value={{
        listingDraft,
        saveListingDraft,
      }}
    >
      {children}
    </ServiceContext.Provider>
  );
}


<!-- demo-store-context分割ユーザー部 -->
import { createContext, useContext } from 'react';
import type { SessionUser } from '@/domain/models';
export interface DemoStoreValue {
  activeUser: SessionUser | null;
  login(email: string, password: string): Promise<void>;
}
export const DemoStoreContext = createContext<DemoStoreValue | null>(null);
export function useDemoStore(): DemoStoreValue {
  const value = useContext(DemoStoreContext);
  if (!value) {
    throw new Error(
      'useDemoStore must be used within DemoStoreProvider',
    );
  }
  return value;
}


<!-- //demo-store-context分割サービス部 -->
import { createContext, useContext } from 'react';
import type { CreateProductInput } from '@/domain/models';
export interface ServiceValue {
  listingDraft: CreateProductInput | null;
  saveListingDraft(draft: CreateProductInput | null): void;
}
export const ServiceContext = createContext<ServiceValue | null>(null);
export function useService(): ServiceValue {
  const value = useContext(ServiceContext);
  if (!value) {
    throw new Error(
      'useService must be used within ServiceProvider',
    );
  }
  return value;
}
